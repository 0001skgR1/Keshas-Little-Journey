function readXML() {
var xhttp = new XMLHttpRequest();
xhttp.onreadystatechange = function() {
if (this.readyState == 4 && this.status == 200) {
var xmlDoc = this.responseXML;
var procoList = xmlDoc.getElementsByTagName("proco");
                    
for (var i = 0; i < procoList.length; i++) {
var name = procoList[i].getElementsByTagName("name")[0].childNodes[0].nodeValue;
var text = procoList[i].getElementsByTagName("text")[0].childNodes[0].nodeValue;
var bgid = procoList[i].getElementsByTagName("bgid")[0].childNodes[0].nodeValue;

console.log("Name: " + name);
console.log("Text: " + text);
console.log("Bgid: " + bgid);
}
}
};
xhttp.open("GET", "data.xml", true);
xhttp.send();
}
